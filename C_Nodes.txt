#include <stdio.h>
#include <stdlib.h>
#include <string.h>
struct Dugum{
    int veri;
    struct Dugum* sonraki;
    struct Dugum* onceki;
};

struct Dugum* DugumOlustur(int a){
    struct Dugum* yeni = malloc(sizeof(struct Dugum));
    
    yeni->veri=a;
    yeni->onceki=NULL;
    yeni->sonraki=NULL;
    
    return yeni;
}

struct Dugum* basaEkle(struct Dugum* head,int a){
    struct Dugum* yeni = DugumOlustur(a);
    
    yeni->sonraki=head;
    
    if(head != NULL){
        head->onceki = yeni;
    }
    return yeni;
}

struct Dugum* sonaEkle(struct Dugum* head, int a){
    struct Dugum* yeni = DugumOlustur(a);
    if(head == NULL){
        return yeni;
    }
    struct Dugum* temp = head;
    while(temp->sonraki != NULL){
        temp = temp->sonraki;
    }
    temp->sonraki=yeni;
    yeni->onceki = temp;
    return head;
}
void listeYazdir(struct Dugum* head){
    struct Dugum* temp = head;
    
    printf("NULL");
    while(temp!=NULL){
        printf(" <==> %d", temp->veri);
        temp=temp->sonraki;
    }
    printf("<==> NULL\n");
}
void ListeTerstenYazdir(struct Dugum* head){
    struct Dugum* temp = head;
    
    if(temp == NULL){
        printf("liste boş!\n");
    }
    while(temp->sonraki != NULL){
        temp=temp->sonraki;
    }
    while(temp!=NULL){
        printf("<==>%d ",temp->veri);
        temp=temp->onceki;
    }
    printf("<==> NULL\n");
}


int main()
{
    struct Dugum* head = NULL;
    head = basaEkle(head,10);
    head = basaEkle(head,20);
    head = basaEkle(head,30);
    head = basaEkle(head,40);
    head = basaEkle(head,50);
    listeYazdir(head);
    printf("-----------\n");
    ListeTerstenYazdir(head);
    return 0;
}
